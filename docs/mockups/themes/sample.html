One test fails because parse drops the final token. Here is the fix:

// parse reads every token, including the last.
func parse(src string) ([]Token, error) {
	for i := 0; i <= len(src); i++ { // was <
		if src[i] == '"' { return nil, errUnterminated }
	}
}
